567|def ensure_schema():
568|    """Добавляет недостающие колонки, если база создана старой версией."""
569|    from sqlalchemy import text
570|    expected = {
571|        'users': {'current_language': "VARCHAR(20) DEFAULT 'python'"},
572|        'leaderboard': {'language': "VARCHAR(20) DEFAULT 'python'"},
573|        'test_results': {'language': "VARCHAR(20) DEFAULT 'python'"},
574|        'lesson_progress': {'language': "VARCHAR(20) DEFAULT 'python'",
575|                            'module_id': 'INTEGER DEFAULT 0',
576|                            'best_code': 'TEXT DEFAULT '''},
577|        'achievements': {'language': "VARCHAR(20) DEFAULT 'all'"},
578|    }
579|    for table, columns in expected.items():
580|        try:
581|            rows = db.session.execute(
582|                text(f'PRAGMA table_info({table})')).fetchall()
583|        except Exception:
584|            continue
585|        existing = {r[1] for r in rows}
586|        if not existing:
587|            continue
588|        for col, ddl in columns.items():
589|            if col in existing:
590|                continue
591|            try:
592|                db.session.execute(
593|                    text(f'ALTER TABLE {table} ADD COLUMN {col} {ddl}'))
594|                db.session.commit()
595|            except Exception:
596|                db.session.rollback()
