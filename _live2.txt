 149| def get_lesson_progress(user, lang_code):
 150|     """Возвращает прогресс пользователя по урокам языка"""
 151|     from models import LessonProgress
 152|     progress = LessonProgress.query.filter_by(
 153|         user_id=user.id,
 154|         language=lang_code
 155|     ).all()
 156| 
 157|     completed_ids = {p.lesson_id for p in progress if p.completed}
 158|     total = get_total_lessons(lang_code)
 159| 
 160|     # следующий урок: первый непройденный, либо самый первый урок курса
 161|     flat = get_flat_lessons(lang_code)
 162|     next_lesson = None
 163|     for lsn in flat:
 164|         if lsn['id'] not in completed_ids:
 165|             next_lesson = lsn
 166|             break
 167|     if next_lesson is None and flat:
 168|         next_lesson = flat[-1]
 169| 
 170|     return {
 171|         'completed': len(completed_ids),
 172|         'total': total,
 173|         'percent': round(len(completed_ids) / total * 100) if total > 0 else 0,
 174|         'completed_ids': completed_ids,
 175|         'next_lesson': next_lesson['id'] if next_lesson else None,
 176|         'next_lesson_title': next_lesson['title'] if next_lesson else None,
 177|         'finished': bool(total) and len(completed_ids) >= total
 178|     }

 180| def get_module_progress(user, lang_code, module_id):
 181|     """Возвращает прогресс по конкретному модулю"""
 182|     from models import LessonProgress
 183|     module = get_module(lang_code, module_id)
 184|     lessons = module.get('lessons', []) if module else []
 185|     progress = LessonProgress.query.filter_by(
 186|         user_id=user.id, language=lang_code,
 187|         module_id=module_id
 188|     ).all()
 189|     by_id = {p.lesson_id: p for p in progress}
 190|     items = []
 191|     done = 0
 192|     for lsn in lessons:
 193|         p = by_id.get(lsn['id'])
 194|         items.append({'lesson': lsn, 'completed': bool(p and p.completed),
 195|                       'attempts': p.attempts if p else 0})
 196|         if p and p.completed:
 197|             done += 1
 198|     total = len(lessons)
 199|     percent = round(done / total * 100) if total else 0
 200|     return {'items': items, 'done': done, 'total': total,
 201|             'percent': percent, 'unlocked': True}

   8| def _run_python_code(code, timeout=5):
   9|     """Execute code in subprocess with full UTF-8 support."""
  10|     temp_path = None
  11|     try:
  12|         with tempfile.NamedTemporaryFile(
  13|             mode='w', suffix='.py', delete=False, encoding='utf-8'
  14|         ) as f:
  15|             f.write(code)
  16|             temp_path = f.name
  17| 
  18|         env = os.environ.copy()
  19|         env['PYTHONIOENCODING'] = 'utf-8'
  20| 
  21|         result = subprocess.run(
  22|             [sys.executable, temp_path],
  23|             capture_output=True, text=True, encoding='utf-8',
  24|             timeout=timeout, cwd=os.path.dirname(temp_path), env=env
  25|         )
  26|         if temp_path and os.path.exists(temp_path):
  27|             os.unlink(temp_path)
  28| 
  29|         return {
  30|             'success': result.returncode == 0,
  31|             'returncode': result.returncode,
  32|             'stdout': result.stdout,
  33|             'stderr': result.stderr
  34|         }
  35|     except subprocess.TimeoutExpired:
  36|         if temp_path and os.path.exists(temp_path):
  37|             try:
  38|                 os.unlink(temp_path)
  39|             except OSError:
  40|                 pass
  41|         return {
  42|             'success': False, 'returncode': None,
  43|             'stdout': '',
  44|             'stderr': 'Превышено время выполнения (5 секунд).'
  45|         }
  46|     except Exception as e:
  47|         if temp_path and os.path.exists(temp_path):
  48|             try:
  49|                 os.unlink(temp_path)
  50|             except OSError:
  51|                 pass
  52|         return {
  53|             'success': False, 'returncode': None,
  54|             'stdout': '',
  55|             'stderr': 'Ошибка при выполнении: ' + str(e)
  56|         }
  57|

  59| def _check_contains(code, check):
  60|     """Проверка для JS/Java/C++: в коде должны присутствовать конструкции."""
  61|     require = check.get('require') or []
  62|     reject = check.get('reject') or []
  63|     missing = [p for p in require if p.lower() not in code.lower()]
  64|     bad = [p for p in reject if p.lower() in code.lower()]
  65|     if missing:
  66|         return {
  67|             'success': False,
  68|             'stdout': '',
  69|             'stderr': 'Не хватает ключевых конструкций: '
  70|                       + ', '.join(f'«{m}»' for m in missing),
  71|             'missing': missing
  72|         }
  73|     if bad:
  74|         return {
  75|             'success': False,
  76|             'stdout': '',
  77|             'stderr': 'Лишние конструкции (попробуй упростить): '
  78|                       + ', '.join(f'«{b}»' for b in bad),
  79|             'bad': bad
  80|         }
  81|     return {
  82|         'success': True,
  83|         'stdout': 'Код содержит все необходимые конструкции. Отлично!',
  84|         'stderr': ''
  85|     }
  86| 
  87| 
  88| def check_code(user_code, check, timeout=5):
  89|     """Проверяет решение по правилам задания.
  90| 
  91|     check — объект из задачи:
  92|       {'type': 'python', 'test_code': '...'}   — реальный запуск Python
  93|       {'type': 'contains', 'require': [...], 'reject': [...]} — для JS/Java/C++
  94|     """
  95|     if not isinstance(check, dict):
  96|         # старая сигнатура: check_code(user_code, test_code)
  97|         check = {'type': 'python', 'test_code': check}
  98|     ctype = check.get('type', 'python')
  99|     if ctype == 'contains':
 100|         return _check_contains(user_code, check)
 101|     full_code = user_code + '\n' + (check.get('test_code') or '')
 102|     return _run_python_code(full_code, timeout)
 103|

  55| def level_progress(xp):
  56|     """Возвращает (xp_накоплено_в_текущем_уровне, xp_нужно_до_следующего)."""
  57|     left = xp
  58|     level = 0
  59|     need = 100
  60|     while left >= need:
  61|         left -= need
  62|         level += 1
  63|         need = 100 * (level + 1) ** 2
  64|     return left, need

 258|             o.type = 'sine';
 259|             o.frequency.setValueAtTime(freq, t);
 260|             o.frequency.exponentialRampToValueAtTime(freq * 0.75, t + 0.1);
 261|             g.gain.setValueAtTime(0.001, t);
 262|             g.gain.exponentialRampToValueAtTime(type === 'ghost' ? 0.04 : 0.06, t + 0.01);
 263|             g.gain.exponentialRampToValueAtTime(0.0001, t + 0.13);
 264|             o.connect(g);
 265|             g.connect(master);
 266|             o.start(t);
 267|             o.stop(t + 0.15);
 268|         } catch (e) { /* звук некритичен */ }
 269|     }
 270| // одна «нота» пада: два осциллятора + фильтр, медленное плавное начало
 271|     function buildVoice(t, freq, det, wave, vol, dur) {
 272|         var o = ctx.createOscillator();
 273|         var g = ctx.createGain();
 274|         var f = ctx.createBiquadFilter();
 275|         o.type = wave;
 276|         o.frequency.setValueAtTime(freq * det, t);
 277|         f.type = 'lowpass';
 278|         f.frequency.setValueAtTime(900, t);
 279|         f.Q.setValueAtTime(0.7, t);
 280|         g.gain.setValueAtTime(0.0001, t);
 281|         g.gain.exponentialRampToValueAtTime(vol, t + 2.2);
 282|         g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
 283|         o.connect(g);
 284|         g.connect(f);
 285|         f.connect(musicGain);
 286|         o.onended = function() {
 287|             try { g.disconnect(); } catch (e2) {}
 288|             try { f.disconnect(); } catch (e2) {}
 289|             var idx = musicNodes.indexOf(o);
 290|             if (idx > -1) musicNodes.splice(idx, 1);
 291|         };
 292|         o.start(t);
 293|         o.stop(t + dur + 0.5);
 294|         return o;
 295|     }

 297|     function scheduleChord() {
 298|         if (!musicOn || !ctx) return;
 299|         var t = Math.max(nextChordTime, ctx.currentTime + 0.02);
 300|         var chord = CHORDS[chordIndex % CHORDS.length];
 301|         chordIndex++;
 302|         var dur = 7.5;
 303| 
 304|         for (var i = 0; i < chord.length; i++) {
 305|             var freq = chord[i];
 306|             musicNodes.push(buildVoice(t, freq, 1.0, 'sine', 0.045, dur));
 307|             musicNodes.push(buildVoice(t, freq, 1.003, 'triangle', 0.026, dur));
 308|         }
 309| 
 310|         nextChordTime = t + dur - 0.6;
 311|         schedulerTimer = setTimeout(scheduleChord,
 312|             Math.max(50, (nextChordTime - ctx.currentTime) * 1000));
 313|     }

 315|     function startMusic() {
 316|         if (!musicOn || !ensureCtx()) return;
 317|         if (schedulerTimer) return;
 318|         nextChordTime = ctx.currentTime + 0.15;
 319|         scheduleChord();
 320|     }

 322|     function stopMusic() {
 323|         if (schedulerTimer) {
 324|             clearTimeout(schedulerTimer);
 325|             schedulerTimer = null;
 326|         }
 327|         var t = ctx ? ctx.currentTime : 0;
 328|         if (ctx && musicGain) {
 329|             musicGain.gain.cancelScheduledValues(t);
 330|             musicGain.gain.setValueAtTime(Math.max(musicGain.gain.value || 0.13, 0.0001), t);
 331|             musicGain.gain.exponentialRampToValueAtTime(0.0001, t + 0.6);
 332|         }
 333|         var nodes = musicNodes.slice();
 334|         musicNodes = [];
 335|         nodes.forEach(function(n) {
 336|             try { n.stop(t + 0.1); } catch (e2) {}
 337|             try { n.disconnect(); } catch (e2) {}
 338|         });
 339|     }
