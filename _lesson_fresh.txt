{% block extra_js %}
<script>
(function() {
    var resetBtn = document.getElementById('resetBtn');
    var runBtn = document.getElementById('runBtn');
    var editor = document.getElementById('codeEditor');
    var output = document.getElementById('outputBox');
    var INITIAL = {{ initial_code | tojson }};
    var lastSolution = {{ (solution or '') | tojson }};

    resetBtn.addEventListener('click', function() {
        editor.value = INITIAL;
        output.style.display = 'none';
        // перерисовать подсветку и высоту редактора после сброса
        editor.dispatchEvent(new Event('input'));
    });

    // экранируем вывод, чтобы <, > и html не ломали результат
    function esc(s) {
        return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
    }

    runBtn.addEventListener('click', function() {
        runBtn.disabled = true;
        runBtn.innerHTML = '<span class="spinner-border spinner-border-sm me-1"></span> Выполнение...';
        output.style.display = 'block';
        output.innerHTML = '<span class="term-title">// РЕЗУЛЬТАТ</span>\nПроверяем код…';

        fetch('/lesson/{{ lesson.id }}', {
            method: 'POST',
            headers: {'Content-Type': 'application/x-www-form-urlencoded'},
            body: 'code=' + encodeURIComponent(editor.value)
        })
        .then(function(r) {
            var ct = r.headers.get('Content-Type') || '';
            if (!ct.includes('application/json')) {
                throw new Error('session');
            }
            return r.json();
        })
        .then(function(data) {
            var html = '<span class="term-title">// РЕЗУЛЬТАТ</span>';
            if (data.stdout) {
                html += '\n<span class="t-out">Output:</span>\n' + esc(data.stdout) + '\n';
            }
            if (data.stderr) {
                html += '\n<span class="t-err">Error:</span>\n' + esc(data.stderr) + '\n';
            }
            if (data.hint) {
                html += '\n<span class="t-hint">💡 Подсказка:</span> ' + esc(data.hint) + '\n';
            }
            if (data.success) {
                html += '\n<span class="t-ok">✅ Урок пройден, отлично!</span>';
                if (data.new_achievements && data.new_achievements.length) {
                    html += '\n<span class="t-hint">🏆 Новое достижение: ' + esc(data.new_achievements.join(', ')) + '</span>';
                }
                if (typeof data.level === 'number') {
                    html += '\n<span class="t-out">Уровень: ' + data.level + ' · XP: ' + data.xp + '</span>';
                }
            } else {
                html += '\n<span class="t-err">❌ Проверка не пройдена — попробуй ещё раз</span>';
            }
            output.innerHTML = html;

            // прогресс подсказок: с 3-й ошибки — лампочка, с 4-й — решение
            if (data.success) {
                document.getElementById('hintBox').style.display = 'none';
                document.getElementById('solutionBox').style.display = 'none';
            } else if (typeof data.attempts === 'number') {
                if (data.attempts >= 3 && data.hint) {
                    document.getElementById('hintBox').style.display = 'block';
                    document.getElementById('hintText').textContent = data.hint;
                }
                if (data.attempts >= 4 && data.solution) {
                    lastSolution = data.solution;
                    document.getElementById('solutionBox').style.display = 'block';
                    document.getElementById('solutionCode').textContent = data.solution;
                }
            }

            runBtn.disabled = false;
            runBtn.innerHTML = '<i class="bi bi-play-circle"></i> Проверить задание';
        })
        .catch(function(err) {
            var msg = err.message === 'session'
                ? 'Сессия истекла — обнови страницу и войди заново.'
                : err.message;
            output.innerHTML = '<span class="term-title">// РЕЗУЛЬТАТ</span>\n<span class="t-err">Ошибка: ' + msg + '</span>';
            runBtn.disabled = false;
            runBtn.innerHTML = '<i class="bi bi-play-circle"></i> Проверить задание';
        });
    });

    // кнопка "Вставить в редактор" под полным решением
    document.getElementById('applySolution').addEventListener('click', function() {
        if (!lastSolution) return;
        editor.value = lastSolution;
        editor.dispatchEvent(new Event('input'));
        editor.focus();
    });

    // Мини-тест урока: мгновенная проверка ответа с объяснением
    {% if lesson.test %}
    var testAnswers = {{ lesson.test | map(attribute='correct') | list | tojson }};
    var testExplains = [];
    {% if lesson.test %}
    {% for tq in lesson.test %}
    testExplains.push({{ tq.explain | default('') | tojson }});
    {% endfor %}
    {% endif %}
    var miniRadios = document.querySelectorAll('.mini-test-opt');
    miniRadios.forEach(function(radio) {
        radio.addEventListener('change', function() {
            var name = radio.name; // lt0, lt1...
            var idx = parseInt(name.replace('lt', ''), 10);
            var box = document.getElementById('ltRes' + idx);
            var chosen = parseInt(radio.value, 10);
            if (chosen === testAnswers[idx]) {
                box.innerHTML = '<span class="t-ok">✅ Правильно!</span> ' +
                    (testExplains[idx] ? '<small class="text-muted">' + esc(testExplains[idx]) + '</small>' : '');
            } else {
                box.innerHTML = '<span class="t-err">❌ Неправильно. Попробуй ещё раз.</span>';
            }
        });
    });
    {% endif %}
})();
</script>
{% endblock %}
