  71| def recalc_level(user):
  72|     """Уровень растёт вместе с XP (та же формула, что в прогресс-баре).
  73|     Уровень никогда не понижается: тест может дать стартовый уровень."""
  74|     xp = user.xp
  75|     level = 0
  76|     need = 100
  77|     while xp >= need:
  78|         xp -= need
  79|         level += 1
  80|         need = 100 * (level + 1) ** 2
  81|     if level > user.level:
  82|         user.level = level
  83|     return user.level

  86| def grant_achievements(user, language='python'):
  87|     """Проверяет условия достижений и выдаёт те, что выполнены.
  88|     Возвращает список новых достижений (каждое даёт бонусный XP)."""
  89|     achieved_ids = {ua.achievement_id for ua in
  90|                     UserAchievement.query.filter_by(user_id=user.id)}
  91|     completed = LessonProgress.query.filter_by(
  92|         user_id=user.id, language=language, completed=True).count()
  93|     has_test = TestResult.query.filter_by(
  94|         user_id=user.id, language=language).count() > 0
  95|     first_try = LessonProgress.query.filter_by(
  96|         user_id=user.id, language=language, completed=True, attempts=1).count() > 0
  97|     total_lessons = get_total_lessons(language)
  98| 
  99|     new = []
 100|     for ach in Achievement.query.all():
 101|         if ach.id in achieved_ids:
 102|             continue
 103|         if ach.language != 'all' and ach.language != language:
 104|             continue
 105|         ct = ach.condition_type or ''
 106|         cv = ach.condition_value or 1
 107|         got = False
 108|         if ct == 'register':
 109|             got = True
 110|         elif ct == 'test':
 111|             got = has_test
 112|         elif ct == 'lesson':
 113|             got = completed >= 1
 114|         elif ct == 'all_lessons':
 115|             got = completed >= total_lessons
 116|         elif ct == '10_correct':
 117|             got = completed >= cv
 118|         elif ct == 'first_try':
 119|             got = first_try
 120|         elif ct.startswith('streak_'):
 121|             got = user.streak_count >= cv
 122|         if not got:
 123|             continue
 124|         db.session.add(UserAchievement(user_id=user.id,
 125|                                        achievement_id=ach.id))
 126|         user.xp += ach.xp_reward or 0
 127|         new.append(ach)
 128|     recalc_level(user)
 129|     db.session.commit()
 130|     return new

 149| def get_lesson_progress(user, lang_code):
 150|     """Возвращает прогресс пользователя по урокам языка"""
 151|     from models import LessonProgress

 153|         user_id=user.id,
 154|         language=lang_code
 155|     ).all()
 156|
