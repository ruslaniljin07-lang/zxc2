 142| {% block extra_js %}
 143| <script>
 144| (function() {
 145|     var resetBtn = document.getElementById('resetBtn');
 146|     var runBtn = document.getElementById('runBtn');
 147|     var editor = document.getElementById('codeEditor');
 148|     var output = document.getElementById('outputBox');
 149|     var INITIAL = {{ initial_code | tojson }};
 150|     var lastSolution = {{ (solution or '') | tojson }};
 151| 
 152|     resetBtn.addEventListener('click', function() {
 153|         editor.value = INITIAL;
 154|         output.style.display = 'none';
 155|         // перерисовать подсветку и высоту редактора после сброса
 156|         editor.dispatchEvent(new Event('input'));
 157|     });
 158| 
 159|     // экранируем вывод, чтобы <, > и html не ломали результат
 160|     function esc(s) {
 161|         return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
 162|     }
 163| 
 164|     runBtn.addEventListener('click', function() {
 165|         runBtn.disabled = true;
 166|         runBtn.innerHTML = '<span class="spinner-border spinner-border-sm me-1"></span> Выполнение...';
 167|         output.style.display = 'block';
 168|         output.innerHTML = '<span class="term-title">// РЕЗУЛЬТАТ</span>\nПроверяем код…';
 169| 
 170|         fetch('/lesson/{{ lesson.id }}', {
 171|             method: 'POST',
 172|             headers: {'Content-Type': 'application/x-www-form-urlencoded'},
 173|             body: 'code=' + encodeURIComponent(editor.value)
 174|         })
 175|         .then(function(r) {
 176|             var ct = r.headers.get('Content-Type') || '';
 177|             if (!ct.includes('application/json')) {
 178|                 throw new Error('session');
 179|             }
 180|             return r.json();
 181|         })
 182|         .then(function(data) {
 183|             var html = '<span class="term-title">// РЕЗУЛЬТАТ</span>';
 184|             if (data.stdout) {
 185|                 html += '\n<span class="t-out">Output:</span>\n' + esc(data.stdout) + '\n';
 186|             }
 187|             if (data.stderr) {
 188|                 html += '\n<span class="t-err">Error:</span>\n' + esc(data.stderr) + '\n';
 189|             }
 190|             if (data.hint) {
 191|                 html += '\n<span class="t-hint">💡 Подсказка:</span> ' + esc(data.hint) + '\n';
 192|             }
 193|             if (data.success) {
 194|                 html += '\n<span class="t-ok">✅ Урок пройден, отлично!</span>';
 195|                 if (data.new_achievements && data.new_achievements.length) {
 196|                     html += '\n<span class="t-hint">🏆 Новое достижение: ' + esc(data.new_achievements.join(', ')) + '</span>';
 197|                 }
 198|                 if (typeof data.level === 'number') {
 199|                     html += '\n<span class="t-out">Уровень: ' + data.level + ' · XP: ' + data.xp + '</span>';
 200|                 }
 201|             } else {
 202|                 html += '\n<span class="t-err">❌ Проверка не пройдена — попробуй ещё раз</span>';
 203|             }
 204|             output.innerHTML = html;
 205| 
 206|             // прогресс подсказок: с 3-й ошибки — лампочка, с 4-й — решение
 207|             if (data.success) {
 208|                 document.getElementById('hintBox').style.display = 'none';
 209|                 document.getElementById('solutionBox').style.display = 'none';
 210|             } else if (typeof data.attempts === 'number') {
 211|                 if (data.attempts >= 3 && data.hint) {
 212|                     document.getElementById('hintBox').style.display = 'block';
 213|                     document.getElementById('hintText').textContent = data.hint;
 214|                 }
 215|                 if (data.attempts >= 4 && data.solution) {
 216|                     lastSolution = data.solution;
 217|                     document.getElementById('solutionBox').style.display = 'block';
 218|                     document.getElementById('solutionCode').textContent = data.solution;
 219|                 }
 220|             }
 221| 
 222|             runBtn.disabled = false;
 223|             runBtn.innerHTML = '<i class="bi bi-play-circle"></i> Проверить задание';
 224|         })
 225|         .catch(function(err) {
 226|             var msg = err.message === 'session'
 227|                 ? 'Сессия истекла — обнови страницу и войди заново.'
 228|                 : err.message;
 229|             output.innerHTML = '<span class="term-title">// РЕЗУЛЬТАТ</span>\n<span class="t-err">Ошибка: ' + msg + '</span>';
 230|             runBtn.disabled = false;
 231|             runBtn.innerHTML = '<i class="bi bi-play-circle"></i> Проверить задание';
 232|         });
 233|     });
 234| 
 235|     // кнопка "Вставить в редактор" под полным решением
 236|     document.getElementById('applySolution').addEventListener('click', function() {
 237|         if (!lastSolution) return;
 238|         editor.value = lastSolution;
 239|         editor.dispatchEvent(new Event('input'));
 240|         editor.focus();
 241|     });
 242| })();
 243| </script>
 244| {% endblock %}
