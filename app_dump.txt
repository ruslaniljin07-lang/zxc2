  43| def recalc_level(user):
  44|     """Уровень растёт вместе с XP (та же формула, что в прогресс-баре).
  45|     Уровень никогда не понижается: тест может дать стартовый уровень."""
  46|     xp = user.xp
  47|     level = 0
  48|     need = 100
  49|     while xp >= need:
  50|         xp -= need
  51|         level += 1
  52|         need = 100 * (level + 1) ** 2
  53|     if level > user.level:
  54|         user.level = level
  55|     return user.level

  58| def grant_achievements(user, language='python'):
  59|     """Проверяет условия достижений и выдаёт те, что выполнены.
  60|     Возвращает список новых достижений (каждое даёт бонусный XP)."""
  61|     achieved_ids = {ua.achievement_id for ua in
  62|                     UserAchievement.query.filter_by(user_id=user.id)}
  63|     completed = LessonProgress.query.filter_by(
  64|         user_id=user.id, language=language, completed=True).count()
  65|     has_test = TestResult.query.filter_by(
  66|         user_id=user.id, language=language).count() > 0
  67|     first_try = LessonProgress.query.filter_by(
  68|         user_id=user.id, language=language, completed=True, attempts=1).count() > 0
  69|     total_lessons = get_total_lessons(language)
  70| 
  71|     new = []
  72|     for ach in Achievement.query.all():
  73|         if ach.id in achieved_ids:
  74|             continue
  75|         if ach.language != 'all' and ach.language != language:
  76|             continue
  77|         ct = ach.condition_type or ''
  78|         cv = ach.condition_value or 1
  79|         got = False
  80|         if ct == 'register':
  81|             got = True
  82|         elif ct == 'test':
  83|             got = has_test
  84|         elif ct == 'lesson':
  85|             got = completed >= 1
  86|         elif ct == 'all_lessons':
  87|             got = completed >= total_lessons
  88|         elif ct == '10_correct':
  89|             got = completed >= cv
  90|         elif ct == 'first_try':
  91|             got = first_try
  92|         elif ct.startswith('streak_'):
  93|             got = user.streak_count >= cv
  94|         if not got:
  95|             continue
  96|         db.session.add(UserAchievement(user_id=user.id,
  97|                                        achievement_id=ach.id))
  98|         user.xp += ach.xp_reward or 0
  99|         new.append(ach)
 100|     recalc_level(user)
 101|     db.session.commit()
 102|     return new
